on_startup = {
	if = {
		limit = {
			NOT = { is_year = 1445 }
		}
		#aet_removeUselessTags = yes 

		if = {
			limit = {
				BUR = { ai = yes }
			}
			aet_burgundianRestoration = yes 
		}
	}
}
on_yearly_pulse = {
	if = {
		limit = {
			ai = no 
			has_country_flag = aet_eliteGuards_commonSetupDone
			NOT = { check_variable = {
				which = aet_eliteGuards_actions_bank
				value = 5
			}}
		}

		change_variable = {
			which = aet_eliteGuards_actions_bank
			value = 1 
		}
	} 
}

on_monthly_pulse = {
	country_event = {
		id = aet_custom_modifier_wrapping.1
		days = 1
	}
	
	country_event = {
		id = aet_custom_modifier_wrapping.1
		days = 8
	}
	
	country_event = {
		id = aet_custom_modifier_wrapping.1
		days = 15
	}
	
	country_event = {
		id = aet_custom_modifier_wrapping.1
		days = 22
	}
}

on_five_year_pulse = {
	if = {
		limit = {
			ai = no 
			aet_eliteGuards_anyEnabled = yes 
		}
		aet_eliteGuards_refreshAll = yes 
	}
}


on_five_year_pulse_2 = {
	if = {
		limit = {
			ai = no 
		}
		every_owned_province = {
			aet_flBuildings_land_forcelimit_removeAllModifiers = yes
			aet_flBuildings_refresh_land_forcelimit = yes
			aet_flBuildings_naval_force_limit_removeAllModifiers = yes
			aet_flBuildings_refresh_naval_forcelimit = yes
		}		
	}
}
